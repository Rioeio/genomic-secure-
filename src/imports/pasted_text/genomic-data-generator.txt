# Generate realistic fake genomic data
class GenomicDataGenerator:
    def __init__(self):
        self.genes = [
            'BRCA1', 'BRCA2', 'TP53', 'EGFR', 'ALK', 'KRAS', 'PIK3CA', 
            'PTEN', 'APC', 'VHL', 'MLH1', 'MSH2', 'MSH6', 'PMS2',
            'ATM', 'CHEK2', 'PALB2', 'RAD51C', 'RAD51D', 'CDH1'
        ]
        
        self.conditions = [
            'Breast Cancer', 'Lung Cancer', 'Colorectal Cancer', 'Prostate Cancer',
            'Ovarian Cancer', 'Pancreatic Cancer', 'Melanoma', 'Lymphoma',
            'Leukemia', 'Brain Tumor', 'Cardiovascular Disease', 'Diabetes Type 2',
            'Alzheimer Disease', 'Parkinson Disease', 'Autoimmune Disease'
        ]
        
        self.populations = ['European', 'African', 'Asian', 'Hispanic', 'Middle Eastern']
        
    def generate_variants(self, num_variants):
        """Generate realistic genomic variants"""
        variants = []
        for i in range(num_variants):
            variant = {
                'id': f'rs{random.randint(1000000, 9999999)}',
                'gene': random.choice(self.genes),
                'chromosome': f'{random.randint(1, 22)}',
                'position': random.randint(1000000, 200000000),
                'reference_allele': random.choice(['A', 'T', 'G', 'C']),
                'alternate_allele': random.choice(['A', 'T', 'G', 'C']),
                'minor_allele_frequency': round(random.uniform(0.001, 0.5), 4),
                'clinical_significance': random.choice([
                    'Pathogenic', 'Likely Pathogenic', 'Uncertain Significance',
                    'Likely Benign', 'Benign'
                ]),
                'impact': random.choice(['High', 'Moderate', 'Low', 'Modifier'])
            }
            variants.append(variant)
        return variants
    
    def generate_patient_genomic_profile(self, patient_id):
        """Generate comprehensive genomic profile for a patient"""
        num_variants = random.randint(5, 25)
        variants = self.generate_variants(num_variants)
        
        return {
            'patient_id': patient_id,
            'variants': variants,
            'polygenic_risk_scores': {
                condition: round(random.uniform(0.1, 0.95), 3)
                for condition in random.sample(self.conditions, 5)
            },
            'pharmacogenomic_markers': {
                'warfarin_sensitivity': random.choice(['High', 'Normal', 'Low']),
                'statin_tolerance': random.choice(['High', 'Normal', 'Low']),
                'plavix_metabolizer': random.choice(['Poor', 'Intermediate', 'Normal', 'Ultra-rapid'])
            },
            'carrier_status': {
                gene: random.choice(['Carrier', 'Non-carrier'])
                for gene in random.sample(['BRCA1', 'BRCA2', 'TP53', 'APC'], 2)
            },
            'ancestry': {
                primary: random.choice(self.populations),
                secondary: random.choice(self.populations),
                admixture_proportion: round(random.uniform(0.1, 0.4), 2)
            }
        }
    
    def generate_clinical_data(self, patient_id):
        """Generate clinical data for patient"""
        return {
            'patient_id': patient_id,
            'age': random.randint(18, 85),
            'gender': random.choice(['Male', 'Female', 'Other']),
            'ethnicity': random.choice(self.populations),
            'diagnoses': random.sample(self.conditions, random.randint(0, 3)),
            'medications': [
                random.choice(['Lisinopril', 'Metformin', 'Atorvastatin', 'Warfarin', 'Plavix'])
                for _ in range(random.randint(0, 3))
            ],
            'lab_results': {
                'glucose': round(random.uniform(70, 200), 1),
                'cholesterol': round(random.uniform(150, 300), 1),
                'blood_pressure_systolic': random.randint(90, 180),
                'blood_pressure_diastolic': random.randint(60, 120)
            },
            'family_history': {
                condition: random.choice(['Present', 'Absent'])
                for condition in random.sample(self.conditions, 3)
            }
        }

    })

@app.route('/api/genomic/genes/<gene_name>', methods=['GET'])
def get_gene_specific_data(gene_name):
    """Get data for specific gene across all patients"""
    gene_variants = []
    gene_stats = {
        'total_patients': 0,
        'variant_count': 0,
        'pathogenic_variants': 0,
        'frequency_distribution': {}
    }
    
    for patient in PATIENTS:
        patient_gene_variants = [
            v for v in patient['genomic_profile']['variants'] 
            if v['gene'] == gene_name
        ]
        
        if patient_gene_variants:
            gene_stats['total_patients'] += 1
            gene_stats['variant_count'] += len(patient_gene_variants)
            
            for variant in patient_gene_variants:
                if variant['clinical_significance'] in ['Pathogenic', 'Likely Pathogenic']:
                    gene_stats['pathogenic_variants'] += 1
                
                # MAF distribution
                maf_range = f"{variant['minor_allele_frequency']:.2f}"
                gene_stats['frequency_distribution'][maf_range] = \
                    gene_stats['frequency_distribution'].get(maf_range, 0) + 1
                
                variant['patient_id'] = patient['id']
                gene_variants.append(variant)
    
    return jsonify({
        "gene": gene_name,
        "statistics": gene_stats,
        "variants": gene_variants,
        "privacy_protected": True
    })

@app.route('/api/genomic/risk-analysis', methods=['POST'])
def genomic_risk_analysis():
    """Perform genomic risk analysis with privacy protection"""
    analysis_params = request.json
    condition = analysis_params.get('condition')
    patient_ids = analysis_params.get('patient_ids', [])
    
    if not patient_ids:
        patient_ids = [p['id'] for p in PATIENTS]
    
    results = []
    for patient_id in patient_ids:
        patient = next((p for p in PATIENTS if p['id'] == patient_id), None)
        if patient:
            risk_score = patient['risk_factors']['genetic_risk_score']
            prs = patient['genomic_profile']['polygenic_risk_scores'].get(condition, 0.5)
            
            results.append({
                'patient_id': patient_id,
                'condition_risk_score': risk_score,
                'polygenic_risk_score': prs,
                'combined_risk': (risk_score + prs) / 2
            })
    
    # Apply differential privacy
    protected_results = pg_agent.apply_dp_to_risk_scores(results, epsilon=0.3)
    
    return jsonify({
        "condition": condition,
        "patient_count": len(protected_results),
        "results": protected_results,
        "privacy_budget_used": 0.3,
        "statistical_significance": "p < 0.05 (protected)"
    })

@app.route('/api/genomic/drug-response', methods=['POST'])
def predict_drug_response():
    """Predict drug response based on genomic markers"""
    drug_params = request.json
    drug = drug_params.get('drug')
    patient_ids = drug_params.get('patient_ids', [])
    
    predictions = []
    for patient_id in patient_ids:
        patient = next((p for p in PATIENTS if p['id'] == patient_id), None)
        if patient:
            # Simple drug response prediction
            base_response = 0.5
            genomic_factors = 0
            
            # Check for relevant variants
            if drug == 'Warfarin':
                markers = patient['genomic_profile']['pharmacogenomic_markers']
                if markers['warfarin_sensitivity'] == 'High':
                    genomic_factors = -0.3
                elif markers['warfarin_sensitivity'] == 'Low':
                    genomic_factors = 0.3
            
            predicted_response = max(0, min(1, base_response + genomic_factors + np.random.normal(0, 0.1)))
            
            predictions.append({
                'patient_id': patient_id,
                'drug': drug,
                'predicted_response': predicted_response,
                'confidence': round(random.uniform(0.7, 0.95), 2),
                'recommendation': 'Standard dosing' if abs(predicted_response - 0.5) < 0.2 else 'Dose adjustment needed'
            })
    
    return jsonify({
        "drug": drug,
        "predictions": predictions,
        "model_type": "Federated Pharmacogenomic Model",
        "accuracy": "87.3% (cross-validated)"
    })

    institution_ids = config.get("institution_ids", [])

            study["progress"] = min(100, study["progress"] + 25)

            total_patients = sum(inst["patients"] for inst in INSTITUTIONS if inst["id"] in institution_ids)
            study["totalPatients"] = int(total_patients * 0.15)  # Assume 15% have genomic data
            study["privacyScore"] = 0.92 + (study["progress"] / 100) * 0.06
            
            # Generate model results
            if study["genomic_focus"]:
                study["biomarker_count"] = len(study["genomic_focus"]) * random.randint(3, 12)
                study["effect_size"] = round(random.uniform(0.2, 0.6), 3)
            break
    
    # Run federated learning with genomic data
    result = fl_agent.run_federated_training(study_id, institution_ids)
    
    # Apply differential privacy
    protected_result = pg_agent.apply_protection(result, config.get("epsilon", 0.5))
    
    return jsonify({
        "study_updated": True,
        "fl_result": result,
        "protected_result": protected_result
    })

# Patient Portal APIs
@app.route('/api/patient/<patient_id>/consent', methods=['POST'])
def update_patient_consent(patient_id):
    consent_data = request.json
    study_id = consent_data.get("study_id")
    action = consent_data.get("action")  # "grant" or "revoke"
    
    # Find patient
    patient = next((p for p in PATIENTS if p["id"] == patient_id), None)
    if not patient:
        return jsonify({"error": "Patient not found"}), 404
    
    # Update consent
    if action == "grant" and study_id not in patient["consentedStudies"]:
        patient["consentedStudies"].append(study_id)
        log_event = {
            "id": f"evt-{len(AUDIT_EVENTS) + 1:03d}",
            "timestamp": datetime.now().isoformat() + "Z",
            "type": "consent_granted",
            "actor": f"patient:{patient_id}",
            "resource": f"study:{study_id}",
            "details": f"Consent granted for genomic study {study_id}"
        }
        AUDIT_EVENTS.append(log_event)
        
    elif action == "revoke" and study_id in patient["consentedStudies"]:
        patient["consentedStudies"].remove(study_id)
        log_event = {
            "id": f"evt-{len(AUDIT_EVENTS) + 1:03d}",
            "timestamp": datetime.now().isoformat() + "Z",
            "type": "consent_revoked",
            "actor": f"patient:{patient_id}",
            "resource": f"study:{study_id}",
            "details": f"Genomic data consent revoked for study {study_id}"
        }
        AUDIT_EVENTS.append(log_event)
    
    return jsonify({"success": True, "patient": patient})

def get_institution_stats(institution_id):
    institution = next((inst for inst in INSTITUTIONS if inst["id"] == institution_id), None)
    if not institution:
        return jsonify({"error": "Institution not found"}), 404
    
    # Calculate genomic-specific stats
    institution_studies = [s for s in STUDIES if institution_id in s["institutions"]]
    privacy_remaining = institution["privacyBudgetTotal"] - institution["privacyBudgetUsed"]
    
    # Count genomic data
    genomic_patients = sum(1 for p in PATIENTS if random.random() < 0.7)  # Simulate 70% have genomic data
    
    return jsonify({
        "institution": institution,
        "studies": institution_studies,
        "privacyRemaining": privacy_remaining,
        "totalPatients": institution["patients"],
        "patientsWithGenomicData": genomic_patients,
        "dataContributions": institution["dataContributions"],
        "genomicMetrics": {
            "sequencing_platform": institution["sequencing_platform"],
            "average_coverage": institution["average_coverage"],
            "data_quality_score": institution["data_quality_score"],
            "total_variants_detected": random.randint(50000, 200000),
            "pathogenic_findings": random.randint(100, 1000)
        }
    })
